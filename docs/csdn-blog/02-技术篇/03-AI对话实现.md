```typescript
class ContextManager {
  private maxTokens = 4000  // 最大上下文 tokens
  
  /**
   * 压缩消息历史
   */
  compressMessages(messages: ChatMessage[]): ChatMessage[] {
    // 1. 保留 system message
    const systemMessage = messages.find(m => m.role === 'system')
    
    // 2. 保留最近的 N 条消息
    const recentMessages = messages.slice(-10)
    
    // 3. 估算 token 数量
    const estimatedTokens = this.estimateTokens(recentMessages)
    
    // 4. 如果超出限制，进一步压缩
    if (estimatedTokens > this.maxTokens) {
      return this.summarizeHistory(messages)
    }
    
    return [systemMessage!, ...recentMessages]
  }
  
  /**
   * 估算 token 数量
   */
  private estimateTokens(messages: ChatMessage[]): number {
    const text = messages.map(m => m.content).join(' ')
    // 粗略估算：1 token ≈ 4 个字符
    return Math.ceil(text.length / 4)
  }
  
  /**
   * 总结历史对话
   */
  private async summarizeHistory(messages: ChatMessage[]): Promise<ChatMessage[]> {
    // 使用 AI 总结历史对话
    const summary = await this.summarizeWithAI(messages)
    
    return [
      { role: 'system', content: SYSTEM_PROMPT },
      { role: 'assistant', content: `之前的对话总结：${summary}` },
      ...messages.slice(-5)  // 保留最近 5 条
    ]
  }
}
```

### 工具调用处理

```typescript
class AIService {
  async chat(
    messages: ChatMessage[],
    onToolCall?: (tool: ToolCall) => Promise<string>
  ): Promise<string> {
    // 调用 OpenAI API
    const response = await this.openai.chat.completions.create({
      model: 'gpt-4',
      messages,
      tools,
      tool_choice: 'auto'  // 让 AI 自动决定是否调用工具
    })

    const message = response.choices[0].message

    // 检查是否有工具调用
    if (message.tool_calls && message.tool_calls.length > 0) {
      // 处理工具调用
      const toolResults: ChatMessage[] = []

      for (const toolCall of message.tool_calls) {
        const functionName = toolCall.function.name
        const functionArgs = JSON.parse(toolCall.function.arguments)

        // 执行工具
        const result = await this.executeToolCall(
          functionName,
          functionArgs,
          onToolCall
        )

        // 添加工具结果到消息列表
        toolResults.push({
          role: 'tool',
          tool_call_id: toolCall.id,
          name: functionName,
          content: result
        })
      }

      // 将工具调用和结果添加到消息列表
      messages.push(message)
      messages.push(...toolResults)

      // 再次调用 AI，让它分析工具结果
      return this.chat(messages, onToolCall)
    }

    // 没有工具调用，直接返回回复
    return message.content || ''
  }

  /**
   * 执行工具调用
   */
  private async executeToolCall(
    functionName: string,
    args: any,
    onToolCall?: (tool: ToolCall) => Promise<string>
  ): Promise<string> {
    if (!onToolCall) {
      throw new Error('No tool call handler provided')
    }

    const toolCall: ToolCall = {
      name: functionName,
      arguments: args
    }

    return await onToolCall(toolCall)
  }
}

interface ToolCall {
  name: string
  arguments: any
}
```

## 流式响应

### 为什么需要流式响应？

**非流式响应**：
- 等待完整响应后一次性显示
- 用户体验差（等待时间长）
- 看起来像"卡住了"

**流式响应**：
- 实时显示 AI 的输出
- 用户体验好（像打字机）
- 感觉更快

### 实现流式响应

```typescript
async chatStream(
  messages: ChatMessage[],
  onChunk: (chunk: string) => void,
  onToolCall?: (tool: ToolCall) => Promise<string>
): Promise<void> {
  const stream = await this.openai.chat.completions.create({
    model: 'gpt-4',
    messages,
    tools,
    stream: true  // 启用流式响应
  })

  let fullContent = ''
  let toolCalls: any[] = []

  for await (const chunk of stream) {
    const delta = chunk.choices[0]?.delta

    // 处理内容
    if (delta?.content) {
      fullContent += delta.content
      onChunk(delta.content)  // 实时回调
    }

    // 处理工具调用
    if (delta?.tool_calls) {
      for (const toolCall of delta.tool_calls) {
        if (!toolCalls[toolCall.index]) {
          toolCalls[toolCall.index] = {
            id: toolCall.id,
            type: toolCall.type,
            function: { name: '', arguments: '' }
          }
        }

        if (toolCall.function?.name) {
          toolCalls[toolCall.index].function.name += toolCall.function.name
        }

        if (toolCall.function?.arguments) {
          toolCalls[toolCall.index].function.arguments += toolCall.function.arguments
        }
      }
    }
  }

  // 处理工具调用
  if (toolCalls.length > 0 && onToolCall) {
    const toolResults: ChatMessage[] = []

    for (const toolCall of toolCalls) {
      const functionName = toolCall.function.name
      const functionArgs = JSON.parse(toolCall.function.arguments)

      const result = await this.executeToolCall(
        functionName,
        functionArgs,
        onToolCall
      )

      toolResults.push({
        role: 'tool',
        tool_call_id: toolCall.id,
        name: functionName,
        content: result
      })
    }

    // 添加到消息列表并继续对话
    messages.push({
      role: 'assistant',
      content: fullContent,
      tool_calls: toolCalls
    })
    messages.push(...toolResults)

    // 递归调用，让 AI 分析工具结果
    await this.chatStream(messages, onChunk, onToolCall)
  }
}
```

## Token 优化

### 1. 压缩上下文

```typescript
/**
 * 智能压缩消息历史
 */
function compressMessages(messages: ChatMessage[]): ChatMessage[] {
  // 1. 保留 system message
  const systemMsg = messages.find(m => m.role === 'system')
  
  // 2. 移除中间的普通对话，只保留重要信息
  const importantMessages = messages.filter(m => 
    m.role === 'system' ||
    m.role === 'tool' ||  // 工具调用结果很重要
    m.content.includes('错误') ||  // 错误信息很重要
    m.content.includes('警告')
  )
  
  // 3. 保留最近的 5 条消息
  const recentMessages = messages.slice(-5)
  
  // 4. 合并去重
  const compressed = [
    systemMsg,
    ...importantMessages.slice(1, -5),
    ...recentMessages
  ].filter(Boolean) as ChatMessage[]
  
  return compressed
}
```

### OpenAI Provider

```typescript
class OpenAIProvider implements AIProvider {
  name = 'OpenAI'
  models = ['gpt-3.5-turbo', 'gpt-4', 'gpt-4-turbo']
  
  private client: OpenAI
  
  constructor(apiKey: string) {
    this.client = new OpenAI({ apiKey })
  }
  
  async chat(messages: ChatMessage[], options?: ChatOptions): Promise<string> {
    const response = await this.client.chat.completions.create({
      model: options?.model || 'gpt-4',
      messages,
      temperature: options?.temperature || 0.7,
      max_tokens: options?.maxTokens,
      tools: options?.tools
    })
    
    return response.choices[0].message.content || ''
  }
  
  async chatStream(
    messages: ChatMessage[],
    onChunk: (chunk: string) => void,
    options?: ChatOptions
  ): Promise<void> {
    const stream = await this.client.chat.completions.create({
      model: options?.model || 'gpt-4',
      messages,
      temperature: options?.temperature || 0.7,
      max_tokens: options?.maxTokens,
      tools: options?.tools,
      stream: true
    })
    
    for await (const chunk of stream) {
      const content = chunk.choices[0]?.delta?.content
      if (content) {
        onChunk(content)
      }
    }
  }
}
```

## 完整示例

```typescript
// AI 对话服务完整实现
class AIChatService {
  private provider: AIProvider
  private contextManager: ContextManager
  private cache: ResponseCache
  
  constructor(provider: AIProvider) {
    this.provider = provider
    this.contextManager = new ContextManager()
    this.cache = new ResponseCache()
  }
  
  async chat(
    messages: ChatMessage[],
    onProgress?: (chunk: string) => void
  ): Promise<string> {
    // 1. 检查缓存
    const lastUserMessage = messages[messages.length - 1]
    const cached = this.cache.get(lastUserMessage.content)
    if (cached) {
      return cached
    }
    
    // 2. 压缩上下文
    const compressed = this.contextManager.compressMessages(messages)
    
    // 3. 调用 AI（流式或非流式）
    let response = ''
    
    if (onProgress) {
      await this.provider.chatStream(
        compressed,
        (chunk) => {
          response += chunk
          onProgress(chunk)
        },
        {
          tools: this.getTools(),
          temperature: 0.7
        }
      )
    } else {
      response = await this.provider.chat(compressed, {
        tools: this.getTools(),
        temperature: 0.7
      })
    }
    
    // 4. 缓存结果
    this.cache.set(lastUserMessage.content, response)
    
    return response
  }
  
  private getTools(): Tool[] {
    return [
      {
        type: 'function',
        function: {
          name: 'execute_ssh_command',
          description: '执行 SSH 命令',
          parameters: {
            type: 'object',
            properties: {
              command: { type: 'string' },
              explanation: { type: 'string' },
              risk_level: { type: 'string', enum: ['low', 'medium', 'high'] }
            },
            required: ['command', 'explanation', 'risk_level']
          }
        }
      }
    ]
  }
}
```

实现一个完整的 AI 对话系统需要考虑：
