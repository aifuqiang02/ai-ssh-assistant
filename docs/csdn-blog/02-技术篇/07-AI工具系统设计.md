## 工具注册表

### 注册表实现

```typescript
// tools/registry.ts
export class ToolRegistry {
  private tools = new Map<string, {
    definition: ToolDefinition
    handler: ToolHandler
  }>()

  /**
   * 注册工具
   */
  register(definition: ToolDefinition, handler: ToolHandler): void {
    this.tools.set(definition.name, { definition, handler })
  }

  /**
   * 获取工具定义
   */
  getDefinition(name: string): ToolDefinition | undefined {
    return this.tools.get(name)?.definition
  }

  /**
   * 获取工具处理器
   */
  getHandler(name: string): ToolHandler | undefined {
    return this.tools.get(name)?.handler
  }

  /**
   * 获取所有工具定义
   */
  getAllDefinitions(): ToolDefinition[] {
    return Array.from(this.tools.values()).map(t => t.definition)
  }

  /**
   * 检查工具是否存在
   */
  has(name: string): boolean {
    return this.tools.has(name)
  }

  /**
   * 注销工具
   */
  unregister(name: string): void {
    this.tools.delete(name)
  }
}

// 创建全局注册表
export const toolRegistry = new ToolRegistry()

// 注册所有工具
toolRegistry.register(executeSSHCommandTool, executeSSHCommandHandler)
toolRegistry.register(readFileTool, readFileHandler)
toolRegistry.register(writeFileTool, writeFileHandler)
toolRegistry.register(listDirectoryTool, listDirectoryHandler)
toolRegistry.register(attemptCompletionTool, attemptCompletionHandler)
```

### 转换为 OpenAI 格式

```typescript
// tools/openai-format.ts
export function toOpenAITools(definitions: ToolDefinition[]): any[] {
  return definitions.map(def => ({
    type: 'function',
    function: {
      name: def.name,
      description: def.description,
      parameters: {
        type: 'object',
        properties: def.parameters.reduce((acc, param) => {
          acc[param.name] = {
            type: param.type,
            description: param.description,
            ...(param.enum && { enum: param.enum })
          }
          return acc
        }, {} as any),
        required: def.parameters
          .filter(p => p.required)
          .map(p => p.name)
      }
    }
  }))
}

// 使用
const tools = toOpenAITools(toolRegistry.getAllDefinitions())
const response = await openai.chat.completions.create({
  model: 'gpt-4',
  messages: messages,
  tools: tools,
  tool_choice: 'auto'
})
```

## 工具执行器

### 执行器实现

```typescript
// tools/executor.ts
export class ToolExecutor {
  constructor(private registry: ToolRegistry) {}

  /**
   * 执行工具调用
   */
  async execute(
    toolName: string,
    params: any
  ): Promise<ToolResult> {
    // 1. 验证工具是否存在
    if (!this.registry.has(toolName)) {
      return {
        success: false,
        error: `工具 ${toolName} 不存在`
      }
    }

    // 2. 获取工具定义和处理器
    const definition = this.registry.getDefinition(toolName)!
    const handler = this.registry.getHandler(toolName)!

    // 3. 验证参数
    const validation = this.validateParams(definition, params)
    if (!validation.valid) {
      return {
        success: false,
        error: validation.error
      }
    }

    // 4. 执行工具
    try {
      const result = await handler(params)
      return result
    } catch (error) {
      return {
        success: false,
        error: error.message
      }
    }
  }

  /**
   * 验证参数
   */
  private validateParams(
    definition: ToolDefinition,
    params: any
  ): { valid: boolean; error?: string } {
    // 检查必需参数
    for (const param of definition.parameters) {
      if (param.required && !(param.name in params)) {
        return {
          valid: false,
          error: `缺少必需参数: ${param.name}`
        }
      }
    }

    // 检查参数类型
    for (const [key, value] of Object.entries(params)) {
      const param = definition.parameters.find(p => p.name === key)
      if (!param) continue

      const actualType = typeof value
      if (actualType !== param.type) {
        return {
          valid: false,
          error: `参数 ${key} 类型错误，期望 ${param.type}，实际 ${actualType}`
        }
      }

      // 检查枚举值
      if (param.enum && !param.enum.includes(value as string)) {
        return {
          valid: false,
          error: `参数 ${key} 值必须是 ${param.enum.join(', ')} 之一`
        }
      }
    }

    return { valid: true }
  }

  /**
   * 批量执行工具
   */
  async executeBatch(
    calls: Array<{ name: string; params: any }>
  ): Promise<ToolResult[]> {
    return Promise.all(
      calls.map(call => this.execute(call.name, call.params))
    )
  }
}

// 创建全局执行器
export const toolExecutor = new ToolExecutor(toolRegistry)
```

### 与 AI 集成

```typescript
// components/AIChatSessionWithTools.vue
async function sendMessage(content: string) {
  // 1. 添加用户消息
  messages.push({
    role: 'user',
    content: content
  })

  // 2. 调用 AI API
  const response = await openai.chat.completions.create({
    model: currentModel.value.id,
    messages: messages,
    tools: toOpenAITools(toolRegistry.getAllDefinitions()),
    tool_choice: 'auto'
  })

  const message = response.choices[0].message

  // 3. 处理工具调用
  if (message.tool_calls) {
    for (const toolCall of message.tool_calls) {
      // 执行工具
      const result = await toolExecutor.execute(
        toolCall.function.name,
        JSON.parse(toolCall.function.arguments)
      )

      // 添加工具结果到消息
      messages.push({
        role: 'tool',
        tool_call_id: toolCall.id,
        content: JSON.stringify(result)
      })
    }

    // 4. 继续对话（让 AI 处理工具结果）
    await sendMessage('')
  } else {
    // 5. 显示 AI 回复
    messages.push(message)
  }
}
```

## 性能优化

### 1. 并行执行工具

```typescript
// 串行执行（慢）
const result1 = await toolExecutor.execute('execute_ssh_command', { command: 'ls' })
const result2 = await toolExecutor.execute('execute_ssh_command', { command: 'pwd' })

// 并行执行（快）
const [result1, result2] = await Promise.all([
  toolExecutor.execute('execute_ssh_command', { command: 'ls' }),
  toolExecutor.execute('execute_ssh_command', { command: 'pwd' })
])
```

### 2. 工具结果缓存

```typescript
class CachedToolExecutor extends ToolExecutor {
  private cache = new Map<string, { result: ToolResult; timestamp: number }>()
  private cacheTTL = 60000  // 1 分钟

  async execute(toolName: string, params: any): Promise<ToolResult> {
    // 生成缓存键
    const cacheKey = `${toolName}:${JSON.stringify(params)}`
    
    // 检查缓存
    const cached = this.cache.get(cacheKey)
    if (cached && Date.now() - cached.timestamp < this.cacheTTL) {
      return cached.result
    }

    // 执行工具
    const result = await super.execute(toolName, params)

    // 缓存结果
    this.cache.set(cacheKey, {
      result,
      timestamp: Date.now()
    })

    return result
  }
}
```

### 3. 超时控制

```typescript
async function executeWithTimeout<T>(
  promise: Promise<T>,
  timeout: number
): Promise<T> {
  return Promise.race([
    promise,
    new Promise<T>((_, reject) =>
      setTimeout(() => reject(new Error('执行超时')), timeout)
    )
  ])
}

// 使用
const result = await executeWithTimeout(
  toolExecutor.execute('execute_ssh_command', { command: 'sleep 100' }),
  5000  // 5 秒超时
)
```

### 4. 限流

```typescript
class RateLimitedExecutor extends ToolExecutor {
  private queue: Array<() => Promise<any>> = []
  private running = 0
  private maxConcurrent = 3

  async execute(toolName: string, params: any): Promise<ToolResult> {
    return new Promise((resolve, reject) => {
      this.queue.push(async () => {
        try {
          const result = await super.execute(toolName, params)
          resolve(result)
        } catch (error) {
          reject(error)
        }
      })
      this.processQueue()
    })
  }

  private async processQueue() {
    if (this.running >= this.maxConcurrent || this.queue.length === 0) {
      return
    }

    this.running++
    const task = this.queue.shift()!
    
    try {
      await task()
    } finally {
      this.running--
      this.processQueue()
    }
  }
}
```
